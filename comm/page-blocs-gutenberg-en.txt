    storageReady = false;
  }

  function rememberState(key, value) {
    if (!storageReady) {
      return;
    }
    window.localStorage.setItem(key, value);
  }

  function isSuppressed() {
    if (!storageReady) {
      return false;
    }
    if (window.localStorage.getItem(subscribedKey) === '1') {
      return true;
    }
    var rawValue = window.localStorage.getItem(dismissedKey);
    if (!rawValue) {
      return false;
    }
    var dismissedAt = parseInt(rawValue, 10);
    if (isNaN(dismissedAt)) {
      return false;
    }
    if ((Date.now() - dismissedAt) < hideDays * 24 * 60 * 60 * 1000) {
      return true;
    }
    return false;
  }

  function openPopup() {
    if (shown) {
      return;
    }
    if (closed) {
      return;
    }
    if (isSuppressed()) {
      return;
    }
    shown = true;
    root.removeAttribute('hidden');
    document.body.classList.add('stluth-stage-news-popup-open');
  }

  function closePopup(markSubscribed) {
    if (closed) {
      return;
    }
    closed = true;
    root.setAttribute('hidden', 'hidden');
    document.body.classList.remove('stluth-stage-news-popup-open');
    rememberState(dismissedKey, String(Date.now()));
    if (markSubscribed) {
      rememberState(subscribedKey, '1');
    }
  }

  function maybeOpenFromScroll() {
    if (shown) {
      return;
    }
    if (closed) {
      return;
    }
    var doc = document.documentElement;
    var scrollable = doc.scrollHeight - window.innerHeight;
    if (scrollable <= 0) {
      return;
    }
    var progress = window.scrollY / scrollable;
    if (progress >= scrollThreshold) {
      openPopup();
    }
  }

  window.setTimeout(function(){
    openPopup();
  }, delayMs);

  window.addEventListener('scroll', maybeOpenFromScroll, { passive: true });

  if (window.matchMedia) {
    var finePointer = window.matchMedia('(pointer:fine)').matches;
    if (finePointer) {
      document.addEventListener('mouseout', function(event){
        if (shown) {
          return;
        }
        if (closed) {
          return;
        }
        if (!event) {
          return;
        }
        if (event.clientY > 12) {
          return;
        }
        openPopup();
      });
    }
  }

  closeButtons.forEach(function(button){
    button.addEventListener('click', function(){
      closePopup(false);
    });
  });

  document.addEventListener('keydown', function(event){
    if (!shown) {
      return;
    }
    if (!event) {
      return;
    }
    if (event.key === 'Escape') {
      closePopup(false);
    }
  });

  if (dialog) {
    dialog.addEventListener('click', function(event){
      if (!event) {
        return;
      }
      event.stopPropagation();
    });
  }

  if (form) {
    form.addEventListener('submit', function(){
      closePopup(true);
    });
  }
})();
</script>
